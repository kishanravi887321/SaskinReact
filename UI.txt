## 1. ENHANCED INTERVIEW SETUP

```typescript
// app/interview/[role]/page.tsx
import { useState, useRef } from 'react';
import { useParams } from 'next/navigation';

interface EnhancedInterviewState {
  sessionId: string | null;
  currentQuestion: string;
  questionNumber: number;
  isRecording: boolean;
  isProcessing: boolean;
  transcription: string;
  feedback: any;
  audioEnabled: boolean;
  pipelineActive: boolean;
  status: 'setup' | 'active' | 'completed';
}

export default function EnhancedInterviewPage() {
  const { role } = useParams(); // Gets role from URL (e.g., 'python-developer')
  const [state, setState] = useState<EnhancedInterviewState>({
    sessionId: null,
    currentQuestion: '',
    questionNumber: 1,
    isRecording: false,
    isProcessing: false,
    transcription: '',
    feedback: null,
    audioEnabled: true,
    pipelineActive: false,
    status: 'setup'
  });

  // START ENHANCED INTERVIEW (NEW API)
  const startEnhancedInterview = async (experience: string, industry: string) => {
    try {
      const response = await fetch(`/api/central/enhanced-interview/start/${role}/`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Authorization': `Bearer ${localStorage.getItem('token')}`
        },
        body: JSON.stringify({
          experience: experience,     // Junior, Mid-level, Senior
          industry: industry,         // Technology, Healthcare, etc.
          audio_enabled: true         // Enable VTT → Gemini → TTV pipeline
        })
      });

      const result = await response.json();

      if (result.status === 'success') {
        setState(prev => ({
          ...prev,
          sessionId: result.session_id,
          currentQuestion: result.question,
          status: 'active',
          audioEnabled: result.audio_enabled,
          pipelineActive: result.pipeline_active
        }));

        // Auto-play first question audio if available
        if (result.question_audio_url) {
          playAudio(result.question_audio_url);
        }
      }
    } catch (error) {
      console.error('Failed to start enhanced interview:', error);
    }
  };

  return (
    <div className="enhanced-interview-container">
      <h1>Enhanced Interview: {role?.replace('-', ' ').toUpperCase()}</h1>

      {/* Pipeline Status Indicator */}
      <div className="pipeline-status">
        <span className={`status-badge ${state.pipelineActive ? 'active' : 'inactive'}`}>
          {state.pipelineActive ? '🎤 Audio Pipeline Active' : '📝 Text Mode'}
        </span>
      </div>

      {/* Rest of your interview components... */}
    </div>
  );
}
```

## 2. AUDIO RECORDING INTEGRATION

```typescript
const mediaRecorder = useRef<MediaRecorder | null>(null);
const audioChunks = useRef<Blob[]>([]);

// Record audio response
const startRecording = async () => {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    mediaRecorder.current = new MediaRecorder(stream);
    audioChunks.current = [];

    mediaRecorder.current.ondataavailable = (event) => {
      audioChunks.current.push(event.data);
    };

    mediaRecorder.current.onstop = () => {
      const audioBlob = new Blob(audioChunks.current, { type: 'audio/wav' });
      submitAudioResponse(audioBlob);
    };

    mediaRecorder.current.start();
    setState(prev => ({ ...prev, isRecording: true }));
  } catch (error) {
    console.error('Microphone access failed:', error);
    // Fallback to text input
    promptTextInput();
  }
};

const stopRecording = () => {
  if (mediaRecorder.current?.state === 'recording') {
    mediaRecorder.current.stop();
    setState(prev => ({ ...prev, isRecording: false, isProcessing: true }));
  }
};
```

## 3. SUBMIT AUDIO THROUGH PIPELINE

```typescript
const submitAudioResponse = async (audioBlob: Blob) => {
  try {
    const formData = new FormData();
    formData.append('session_id', state.sessionId!);
    formData.append('audio', audioBlob, 'response.wav');

    const response = await fetch('/api/central/enhanced-interview/submit/', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('token')}`
      },
      body: formData
    });

    const result = await response.json();

    if (result.status === 'success') {
      // PIPELINE RESULTS:
      setState(prev => ({
        ...prev,
        transcription: result.transcription,        // VTT output
        feedback: result.feedback,                  // Gemini AI response
        isProcessing: false
      }));

      // Play AI feedback audio (TTV output)
      if (result.audio_urls?.feedback_audio) {
        setTimeout(() => playAudio(result.audio_urls.feedback_audio), 1000);
      }

      // Handle next question or completion
      if (result.interview_status === 'continue') {
        setTimeout(() => {
          setState(prev => ({
            ...prev,
            currentQuestion: result.next_question,
            questionNumber: result.progress.current_question,
            transcription: '',
            feedback: null
          }));

          // Play next question audio
          if (result.audio_urls?.next_question_audio) {
            playAudio(result.audio_urls.next_question_audio);
          }
        }, 3000);
      } else if (result.interview_status === 'completed') {
        setState(prev => ({ ...prev, status: 'completed' }));

        // Play final summary audio
        if (result.audio_urls?.summary_audio) {
          playAudio(result.audio_urls.summary_audio);
        }
      }
    }
  } catch (error) {
    console.error('Audio processing failed:', error);
    setState(prev => ({ ...prev, isProcessing: false }));
  }
};
```

const submitTextResponse = async (textAnswer: string) => {
  try {
    const formData = new FormData();
    formData.append('session_id', state.sessionId!);
    formData.append('answer', textAnswer);

    const response = await fetch('/api/central/enhanced-interview/submit/', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${localStorage.getItem('token')}`
      },
      body: formData
    });

    const result = await response.json();

    if (result.status === 'success') {
      setState(prev => ({
        ...prev,
        transcription: textAnswer,
        feedback: result.feedback,
        isProcessing: false
      }));
    }
  } catch (error) {
    console.error('Text processing failed:', error);
  }
};
